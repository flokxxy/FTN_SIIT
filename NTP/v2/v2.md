//1 
// This is a common rule. For each rule a class with the same name will be
// created.
Program:
  'def' name=ID '=' '{'
    commands*=Command    // *= operator means zero or more matches.
                         // commands will be
                         // a list of Command objects
  '};'

  'run' '(' fun_name=ID ')' ';'
;

// This is an example of abstract rule. Command class will never be instantiated
// in the model.
Command:
  InitialCommand | MoveCommand
;

InitialCommand:
  'init' '(' x=INT ',' y=INT ')' ';'
;

MoveCommand:
 'move' '(' direction=Direction ','? (steps=INT)? ')' ';'
;

// This is an example of a Match Rule
// Match rules has either string match, regex match or other match rule as
// its alternatives (e.g. INT, STRING... or some user match rule)
// Match rule is treated as a contained match. No class will get created.
Direction:
  "up"|"down"|"left"|"right"
;


// Special rule for comments. In robot programs comments start with //
Comment:
  /\/\/.*$/
;





//что описывает вот это
def p1 = {
 init (3, 1);
 move (up, 4);
 move (left, 9);
 move (down);
 move (right, 1);
};
run(p1);

Program:
  'begin'
    commands*=Command    
  'end'
;


Command:
  InitialCommand | MoveCommand | ForCommand |IfCommand
;

ForCommand:
    'for' var=ID 'in' lower=INT '..' upper=INT
    commands*=Command   
    'end'
;
IfCommand:
    'if' var=ID oper=Operations against=INT
    commands+=Command
    'end'
;
InitialCommand:
  'initial' x=INT ',' y=INT
;

MoveCommand:
  direction=Direction (steps=StepExeption)?
;

StepExeption:
    !Command !'end' (INT|ID)?
;
Direction:
  "up"|"down"|"left"|"right"
;
Operations:
">"|"<"|"=="|"<="|">="|"!="
;


Comment:
  /\/\/.*$/
;

begin
  initial 3, 1
  up 4          
  left 9
  down          
  right 1
  for i in 5..10
    up 2
    right i
    for i in 5..10
        right 1
    end
   end
    if x>5
        right x
    end 
 right
end

    commands*=Command    
  'end'
;


Command:
  InitialCommand | MoveCommand | ForCommand |IfCommand
;

ForCommand:
    'for' var=ID 'in' lower=INT '..' upper=INT
    commands*=Command   
    'end'
;
IfCommand:
    'if' conditions += IfConditions
    commands+=Command
    'end'
;
InitialCommand:
  'initial' x=INT ',' y=INT
;

MoveCommand:
  direction=Direction (steps=StepExeption)?
;
IfConditions:
    BooleanConditions | BooleanConnector
;
BooleanConditions:
    var=ID operator=Operations against=INT
;

BooleanConnector:
    "and"|"or"
;
StepExeption:
    !Command !'end' (INT|ID)?
;
Direction:
  "up"|"down"|"left"|"right"
;
Operations:
">"|"<"|"=="|"<="|">="|"!="
;


Comment:
  /\/\/.*$/
;
